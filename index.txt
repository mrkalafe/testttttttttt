async function loadDashboard() {

    const { data, error } = await db
        .from("articles")
        .select("id, title, category, status, created_at")
        .order("created_at", { ascending: false });

    if (error) {

        console.error("Database error:", error);

        return;

    }

    const articles = data || [];

    const published = articles.filter(
        item => item.status === "published"
    ).length;

    const drafts = articles.filter(
        item => item.status === "draft"
    ).length;


    document.getElementById("totalArticles").textContent =
        articles.length.toLocaleString("fa-IR");

    document.getElementById("publishedArticles").textContent =
        published.toLocaleString("fa-IR");

    document.getElementById("draftArticles").textContent =
        drafts.toLocaleString("fa-IR");


    renderArticles(articles);

}




function renderArticles(articles) {

    const table = document.getElementById("articlesTable");

    table.replaceChildren();

    if (articles.length === 0) {

        const row = document.createElement("tr");
        const cell = document.createElement("td");

        cell.colSpan = 5;
        cell.className = "empty";
        cell.textContent = "هنوز مطلبی ثبت نشده است.";

        row.appendChild(cell);
        table.appendChild(row);

        return;
    }

    articles.slice(0, 5).forEach(function(article) {

        const row = document.createElement("tr");

        const title = document.createElement("td");
        title.textContent = article.title || "بدون عنوان";

        const category = document.createElement("td");
        category.textContent = article.category || "بدون دسته‌بندی";

        const statusCell = document.createElement("td");
        const status = document.createElement("span");

        status.className = "status";

        if (article.status === "published") {
            status.textContent = "منتشرشده";
        } else {
            status.textContent = "پیش‌نویس";
            status.classList.add("draft");
        }

        statusCell.appendChild(status);

        const date = document.createElement("td");

        date.textContent = article.created_at
            ? new Date(article.created_at).toLocaleDateString("fa-IR")
            : "-";

        const actions = document.createElement("td");

        const editButton = document.createElement("button");
        editButton.textContent = "ویرایش";
        editButton.className = "btn";
        editButton.dataset.action = "edit";
        editButton.dataset.id = article.id;

        const deleteButton = document.createElement("button");
        deleteButton.textContent = "حذف";
        deleteButton.className = "btn";
        deleteButton.style.background = "#f5dddd";
        deleteButton.style.marginRight = "5px";
        deleteButton.dataset.action = "delete";
        deleteButton.dataset.id = article.id;

        actions.append(editButton, deleteButton);

        row.append(
            title,
            category,
            statusCell,
            date,
            actions
        );

        table.appendChild(row);

    });
}

async function loadAdminSettings() {
    const button = document.getElementById("refreshAdminSettings");
    button.disabled = true;
    setSettingsStatus("generalSettingsStatus", "در حال دریافت تنظیمات...");
    try {
        const { data, error } = await db.functions.invoke("account-manager", {
            body: { action: "admin_settings_get" }
        });
        if (error) throw error;
        if (!data || data.success !== true) {
            throw new Error(data?.error || "دریافت تنظیمات ناموفق بود.");
        }

        const settings = data.settings || {};
        const general = settings.general || {};
        const membership = settings.membership || {};
        const maintenance = settings.maintenance || {};

        document.getElementById("settingSiteName").value = general.site_name || "";
        document.getElementById("settingSiteDescription").value = general.site_description || "";
        document.getElementById("settingLogoUrl").value = general.logo_url || "";
        document.getElementById("settingRegistrationEnabled").checked = membership.registration_enabled !== false;
        document.getElementById("settingMaintenanceEnabled").checked = maintenance.enabled === true;
        document.getElementById("settingMaintenanceMessage").value = maintenance.message || "";

        setSettingsStatus("generalSettingsStatus", "تنظیمات با موفقیت دریافت شد.");
        setSettingsStatus("membershipSettingsStatus", "");
        setSettingsStatus("maintenanceSettingsStatus", "");
    } catch (error) {
        console.error("Admin settings load failed:", error);
        setSettingsStatus("generalSettingsStatus", error.message || "دریافت تنظیمات انجام نشد.", true);
    } finally {
        button.disabled = false;
    }
}

async function saveAdminSetting(action, payload, statusId, buttonId) {
    const button = document.getElementById(buttonId);
    button.disabled = true;
    setSettingsStatus(statusId, "در حال ذخیره...");
    try {
        const { data, error } = await db.functions.invoke("account-manager", {
            body: { action, ...payload }
        });
        if (error) {
            let detail = error.message || "خطا در ارتباط با سرور";
            if (error.context instanceof Response) {
                try {
                    const responseText = await error.context.clone().text();
                    if (responseText) {
                        const parsed = JSON.parse(responseText);
                        detail = parsed.error || responseText;
                    }
                } catch (_) {}
            }
            throw new Error(detail);
        }
        if (!data || data.success !== true) {
            throw new Error(data?.error || "ذخیره تنظیمات ناموفق بود.");
        }
        setSettingsStatus(statusId, data.message || "با موفقیت ذخیره شد.");
    } catch (error) {
        console.error("Admin setting save failed:", action, error);
        setSettingsStatus(statusId, error.message || "ذخیره تنظیمات انجام نشد.", true);
    } finally {
        button.disabled = false;
    }
}

function articleStatusLabel(status) {
    return status === "published" ? "منتشرشده" : "پیش‌نویس";
}

function renderAdminArticles() {
    const table = document.getElementById("adminArticlesTable");
    const search = document.getElementById("articleSearchInput").value.trim().toLocaleLowerCase();
    const statusFilter = document.getElementById("articleStatusFilter").value;

    const filtered = adminArticlesCache.filter(function(item) {
        const matchesSearch = !search ||
            String(item.title || "").toLocaleLowerCase().includes(search) ||
            String(item.slug || "").toLocaleLowerCase().includes(search);
        const matchesStatus = statusFilter === "all" || (item.status || "draft") === statusFilter;
        return matchesSearch && matchesStatus;
    });

    table.replaceChildren();

    if (!filtered.length) {
        const row = document.createElement("tr");
        const cell = document.createElement("td");
        cell.colSpan = 6;
        cell.className = "empty";
        cell.textContent = adminArticlesCache.length
            ? "مقاله‌ای با این جست‌وجو یا فیلتر پیدا نشد."
            : "هنوز مقاله‌ای ثبت نشده است.";
        row.appendChild(cell);
        table.appendChild(row);
    } else {
        filtered.forEach(function(item) {
            const row = document.createElement("tr");

            const title = document.createElement("td");
            title.textContent = item.title || "بدون عنوان";

            const slug = document.createElement("td");
            slug.textContent = item.slug || "-";
            slug.style.direction = "ltr";
            slug.style.textAlign = "right";

            const category = document.createElement("td");
            category.textContent = item.category || "بدون دسته‌بندی";

            const statusCell = document.createElement("td");
            const badge = document.createElement("span");
            badge.className = "status" + (item.status === "published" ? "" : " draft");
            badge.textContent = articleStatusLabel(item.status);
            statusCell.appendChild(badge);

            const date = document.createElement("td");
            date.textContent = item.created_at
                ? new Date(item.created_at).toLocaleDateString("fa-IR")
                : "-";

            const actionsCell = document.createElement("td");
            const actions = document.createElement("div");
            actions.className = "article-row-actions";

            const edit = document.createElement("button");
            edit.type = "button";
            edit.className = "article-edit";
            edit.textContent = "ویرایش";
            edit.dataset.action = "edit";
            edit.dataset.id = item.id;

            const del = document.createElement("button");
            del.type = "button";
            del.className = "article-delete";
            del.textContent = "حذف";
            del.dataset.action = "delete";
            del.dataset.id = item.id;

            actions.append(edit, del);
            actionsCell.appendChild(actions);
            row.append(title, slug, category, statusCell, date, actionsCell);
            table.appendChild(row);
        });
    }

    document.getElementById("adminArticlesCount").textContent =
        "نمایش " + filtered.length.toLocaleString("fa-IR") +
        " مقاله از " + adminArticlesCache.length.toLocaleString("fa-IR") + " مقاله";
}

async function loadAdminDiscussions() {
    const table = document.getElementById("discussionsAdminTable");
    table.innerHTML = "<tr><td colspan='5'>در حال دریافت بحث‌ها...</td></tr>";

    try {
        const discussionsResult = await db.from("discussions")
            .select("id,name,title,body,category,created_at,is_pinned,is_closed")
            .order("is_pinned", {ascending:false})
            .order("created_at", {ascending:false});

        if (discussionsResult.error) throw discussionsResult.error;
        const data = discussionsResult.data || [];
        if (!data.length) {
            table.innerHTML = "<tr><td colspan='5'>هنوز بحثی ثبت نشده است.</td></tr>";
            return;
        }

        const discussionIds = data.map(item => item.id);
        const [contactsResult, repliesResult] = await Promise.all([
            db.from("discussion_contacts").select("discussion_id,email").in("discussion_id", discussionIds),
            db.from("discussion_replies").select("id,discussion_id,name,body,created_at").in("discussion_id", discussionIds).order("created_at", {ascending:true})
        ]);
        if (contactsResult.error) throw contactsResult.error;
        if (repliesResult.error) throw repliesResult.error;

        const contactsByDiscussion = {};
        (contactsResult.data || []).forEach(contact => {
            contactsByDiscussion[String(contact.discussion_id)] = contact.email;
        });

        const replies = repliesResult.data || [];
        const replyIds = replies.map(reply => reply.id);
        let replyContacts = [];
        if (replyIds.length) {
            const replyContactsResult = await db.from("reply_contacts")
                .select("reply_id,email").in("reply_id", replyIds);
            if (replyContactsResult.error) throw replyContactsResult.error;
            replyContacts = replyContactsResult.data || [];
        }
        const emailByReply = {};
        replyContacts.forEach(contact => { emailByReply[String(contact.reply_id)] = contact.email; });

        const repliesByDiscussion = {};
        replies.forEach(reply => {
            reply.email = emailByReply[String(reply.id)] || "";
            const key = String(reply.discussion_id);
            if (!repliesByDiscussion[key]) repliesByDiscussion[key] = [];
            repliesByDiscussion[key].push(reply);
        });

        table.replaceChildren();
        data.forEach(function(item) {
            const itemReplies = repliesByDiscussion[String(item.id)] || [];
            const tr = document.createElement("tr");
            const topic = document.createElement("td");
            const title = document.createElement("strong"); title.textContent = item.title;
            const body = document.createElement("div"); body.className = "small-muted"; body.textContent = item.body;
            topic.append(title, body);
            const author = document.createElement("td");
            author.textContent = item.name + " | " + (contactsByDiscussion[String(item.id)] || "ایمیل ثبت نشده");
            const date = document.createElement("td"); date.textContent = new Date(item.created_at).toLocaleString("fa-IR");
            const state = document.createElement("td");
            state.textContent = (item.is_pinned ? "📌 سنجاق‌شده · " : "") + (item.is_closed ? "بسته" : "باز");
            const actionsCell = document.createElement("td");
            const actions = document.createElement("div"); actions.className = "moderation-actions";
            const pin = document.createElement("button"); pin.className = "pin-action";
            pin.textContent = item.is_pinned ? "برداشتن سنجاق" : "سنجاق‌کردن";
            pin.onclick = () => updateDiscussion(item.id, {is_pinned:!item.is_pinned});
            const close = document.createElement("button"); close.className = "close-action";
            close.textContent = item.is_closed ? "بازکردن پاسخ‌ها" : "بستن پاسخ‌ها";
            close.onclick = () => updateDiscussion(item.id, {is_closed:!item.is_closed});
            const repliesButton = document.createElement("button"); repliesButton.className = "btn";
            repliesButton.textContent = "پاسخ‌ها (" + itemReplies.length + ")";
            repliesButton.onclick = () => toggleAdminReplies(itemReplies, actionsCell);
            const del = document.createElement("button"); del.className = "delete-action"; del.textContent = "حذف بحث";
            del.onclick = () => deleteDiscussion(item.id);
            actions.append(pin, close, repliesButton, del);
            actionsCell.append(actions);
            tr.append(topic, author, date, state, actionsCell);
            table.append(tr);
        });
    } catch (error) {
        console.error("Admin discussions load error:", error);
        table.innerHTML = "<tr><td colspan='5'>خطا در دریافت اطلاعات: " + escapeAdminText(error.message || error) + "</td></tr>";
    }
}
function escapeAdminText(value) {
    return String(value == null ? "" : value).replace(/[&<>"']/g, c => ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c]));
}

function escapeAdminUserText(value) {
    return String(value == null ? "" : value).replace(/[&<>"']/g, function(character) {
        return {"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[character];
    });
}

function formatAdminUserDate(value) {
    if (!value) return "—";
    const date = new Date(value);
    return Number.isNaN(date.getTime()) ? "—" : date.toLocaleDateString("fa-IR");
}

async function callAccountManager(action, values) {
    const body = Object.assign({action: action}, values || {});
    const result = await db.functions.invoke("account-manager", {body: body});
    if (result.error) {
        let message = result.error.message || "ارتباط با سرویس مدیریت کاربران ناموفق بود.";
        if (result.data && result.data.error) message = result.data.error;
        throw new Error(message);
    }
    const payload = result.data || {};
    if (payload.success === false || payload.ok === false || payload.error) {
        throw new Error(payload.error || payload.message || "درخواست توسط سرور انجام نشد.");
    }
    return payload;
}

function updateAdminUsersPagination(pagination) {
    const page = Number(pagination && pagination.page) || adminUsersCurrentPage;
    const totalPages = Number(pagination && pagination.total_pages) || 1;
    const total = Number(pagination && pagination.total);
    adminUsersCurrentPage = page;
    adminUsersTotalPages = Math.max(1, totalPages);
    document.getElementById("adminUsersPageInfo").textContent =
        "صفحه " + page.toLocaleString("fa-IR") + " از " +
        adminUsersTotalPages.toLocaleString("fa-IR") +
        (Number.isFinite(total) ? " | " + total.toLocaleString("fa-IR") + " کاربر" : "");
    document.getElementById("adminUsersPrev").disabled = page <= 1;
    document.getElementById("adminUsersNext").disabled = page >= adminUsersTotalPages;
}

        statusButton.type = "button";
        statusButton.className = user.is_suspended ? "user-unsuspend-btn" : "user-suspend-btn";
        statusButton.textContent = user.is_suspended ? "رفع تعلیق" : "تعلیق کاربر";
        statusButton.dataset.userAction = user.is_suspended ? "unsuspend" : "suspend";
        statusButton.dataset.userId = userId || "";

        actions.append(detailsButton, statusButton);
        actionsCell.appendChild(actions);
        row.append(username, displayName, createdAt, status, actionsCell);
        table.appendChild(row);
    });
}

async function loadAdminUsers() {
    const requestNumber = ++adminUsersRequestNumber;
    const refreshButton = document.getElementById("refreshAdminUsers");
    const table = document.getElementById("adminUsersTable");
    const search = document.getElementById("adminUsersSearch").value.trim();

    refreshButton.disabled = true;
    refreshButton.textContent = "در حال دریافت...";
    setAdminUsersMessage("", false);
    table.innerHTML = '<tr><td class="empty" colspan="5">در حال دریافت کاربران...</td></tr>';

    try {
        const payload = await callAccountManager("admin_users_list", {
            page: adminUsersCurrentPage,
            limit: adminUsersPageSize,
            search: search
        });
        if (requestNumber !== adminUsersRequestNumber) return;
        const users = Array.isArray(payload.users) ? payload.users : [];
        renderAdminUsers(users);
        updateAdminUsersPagination(payload.pagination || {});
        const total = payload.pagination && Number(payload.pagination.total);
        setAdminUsersMessage(
            Number.isFinite(total)
                ? "تعداد کاربران: " + total.toLocaleString("fa-IR")
                : "فهرست کاربران دریافت شد.",
            false
        );
    } catch (error) {
        if (requestNumber !== adminUsersRequestNumber) return;
        console.error("account-manager admin_users_list:", error);
        table.innerHTML = '<tr><td class="empty" colspan="5">دریافت فهرست کاربران ناموفق بود.</td></tr>';
        setAdminUsersMessage(error.message, true);
    } finally {
        if (requestNumber === adminUsersRequestNumber) {
            refreshButton.disabled = false;
            refreshButton.textContent = "به‌روزرسانی";
        }
    }
}

async function showAdminUserDetails(userId) {
    const panel = document.getElementById("adminUserDetailsPanel");
    const content = document.getElementById("adminUserDetailsContent");
    panel.style.display = "block";
    content.textContent = "در حال دریافت جزئیات کاربر...";
    try {
        const payload = await callAccountManager("admin_user_details", {user_id: userId});
        const user = payload.user || {};
        content.textContent = JSON.stringify(user, null, 2);
        panel.scrollIntoView({behavior:"smooth", block:"nearest"});
    } catch (error) {
        content.textContent = "دریافت جزئیات کاربر ناموفق بود:\n" + error.message;
        setAdminUsersMessage(error.message, true);
    }
}

async function changeAdminUserStatus(userId, suspend) {
    const prompt = suspend
        ? "آیا از تعلیق این حساب کاربری مطمئن هستی؟"
        : "آیا می‌خواهی تعلیق این حساب برداشته شود؟";
    if (!confirm(prompt)) return;

    setAdminUsersMessage("در حال ثبت تغییر وضعیت کاربر...", false);
    try {
        await callAccountManager(
            suspend ? "admin_user_suspend" : "admin_user_unsuspend",
            {user_id: userId}
        );
        setAdminUsersMessage(
            suspend ? "حساب کاربر با موفقیت تعلیق شد." : "تعلیق حساب کاربر برداشته شد.",
            false
        );
        await loadAdminUsers();
    } catch (error) {
        console.error("account-manager user status:", error);
        setAdminUsersMessage(error.message, true);
    }
}

        loadAdminUsers();

    }

});
document.getElementById("adminUsersTable").addEventListener("click", function(event) {
    const button = event.target.closest("button[data-user-action]");
    if (!button) return;
    const userId = button.dataset.userId;
    if (!userId) {
        setAdminUsersMessage("شناسه کاربر در پاسخ سرویس وجود ندارد.", true);
        return;
    }
    if (button.dataset.userAction === "details") {
        showAdminUserDetails(userId);
    } else {
        changeAdminUserStatus(userId, button.dataset.userAction === "suspend");
    }
});
